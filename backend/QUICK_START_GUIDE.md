JWT工具类框架\n\n```java\n@Component\npublic class JwtTokenProvider {\n    \n    @Value(\"${jwt.secret}\")\n    private String secretKey;\n    \n    @Value(\"${jwt.expiration}\")\n    private long tokenExpiration;\n    \n    // 生成token\n    public String generateToken(Long userId, String username, String role) {\n        return Jwts.builder()\n                .setSubject(userId.toString())\n                .claim(\"username\", username)\n                .claim(\"role\", role)\n                .setIssuedAt(new Date())\n                .setExpiration(new Date(System.currentTimeMillis() + tokenExpiration))\n                .signWith(SignatureAlgorithm.HS512, secretKey)\n                .compact();\n    }\n    \n    // 验证token\n    public boolean validateToken(String token) {\n        try {\n            Jwts.parser().setSigningKey(secretKey).parseClaimsJws(token);\n            return true;\n        } catch (JwtException e) {\n            return false;\n        }\n    }\n    \n    // 获取userId\n    public Long getUserIdFromToken(String token) {\n        Claims claims = Jwts.parser()\n                .setSigningKey(secretKey)\n                .parseClaimsJws(token)\n                .getBody();\n        return Long.parseLong(claims.getSubject());\n    }\n}\n```\n\n### 2. AuthService框架\n\n```java\n@Service\n@RequiredArgsConstructor\npublic class AuthService {\n    \n    private final UserRepository userRepository;\n    private final JwtTokenProvider tokenProvider;\n    private final PasswordEncoder passwordEncoder;\n    \n    // 登录\n    public AuthResponse login(String username, String password) {\n        // 查询用户\n        User user = userRepository.findByUsername(username)\n                .orElseThrow(() -> new RuntimeException(\"用户不存在\"));\n        \n        // 验证密码\n        if (!passwordEncoder.matches(password, user.getPassword())) {\n            throw new RuntimeException(\"密码错误\");\n        }\n        \n        // 生成tokens\n        String accessToken = tokenProvider.generateToken(user.getId(), user.getUsername(), user.getRole());\n        String refreshToken = tokenProvider.generateRefreshToken(user.getId());\n        \n        return AuthResponse.builder()\n                .token(accessToken)\n                .refreshToken(refreshToken)\n                .user(UserResponse.from(user))\n                .build();\n    }\n    \n    // 注册\n    public AuthResponse register(RegisterRequest request) {\n        // 检查用户是否已存在\n        if (userRepository.existsByUsername(request.getUsername())) {\n            throw new RuntimeException(\"用户名已存在\");\n        }\n        \n        // 创建用户\n        User user = new User();\n        user.setUsername(request.getUsername());\n        user.setEmail(request.getEmail());\n        user.setPhone(request.getPhone());\n        user.setName(request.getName());\n        user.setPassword(passwordEncoder.encode(request.getPassword()));\n        user.setRole(\"salesman\");\n        \n        userRepository.save(user);\n        \n        // 返回登录响应\n        return login(request.getUsername(), request.getPassword());\n    }\n}\n```\n\n### 3. Controller框架\n\n```java\n@RestController\n@RequestMapping(\"/api/auth\")\n@RequiredArgsConstructor\npublic class AuthController {\n    \n    private final AuthService authService;\n    \n    @PostMapping(\"/login\")\n    public ApiResponse<AuthResponse> login(@Valid @RequestBody LoginRequest request) {\n        return ApiResponse.success(authService.login(request.getUsername(), request.getPassword()));\n    }\n    \n    @PostMapping(\"/register\")\n    public ApiResponse<AuthResponse> register(@Valid @RequestBody RegisterRequest request) {\n        return ApiResponse.success(authService.register(request));\n    }\n    \n    @PostMapping(\"/refresh\")\n    public ApiResponse<AuthResponse> refresh(@RequestBody RefreshRequest request) {\n        return ApiResponse.success(authService.refreshToken(request.getRefreshToken()));\n    }\n    \n    @PostMapping(\"/logout\")\n    @PreAuthorize(\"isAuthenticated()\")\n    public ApiResponse<String> logout() {\n        return ApiResponse.success(\"Logout successful\");\n    }\n    \n    @GetMapping(\"/verify\")\n    @PreAuthorize(\"isAuthenticated()\")\n    public ApiResponse<Object> verify() {\n        return ApiResponse.success(Map.of(\n            \"valid\", true,\n            \"expiresIn\", 3600\n        ));\n    }\n}\n```\n\n### 4.
